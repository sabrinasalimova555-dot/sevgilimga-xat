      reset(initial = false) {

        this.x = Math.random() * width;

        this.y = initial ? Math.random() * height : -20;

        this.size = Math.random() * 9 + 6;

        this.speedY = Math.random() * 1.1 + 0.45;

        this.speedX = Math.random() * 0.8 - 0.4;

        this.angle = Math.random() * Math.PI * 2;

        this.spin = (Math.random() - 0.5) * 0.03;

        this.type = Math.random() > 0.32 ? 'petal' : 'heart';

        this.opacity = Math.random() * 0.45 + 0.25;

        const colors = ['#f4bac6', '#e89aa9', '#fad4dc', '#d98290'];

        this.color = colors[Math.floor(Math.random() * colors.length)];

      }


      update() {

        this.y += this.speedY;

        this.x += this.speedX + Math.sin(this.angle) * 0.4;

        this.angle += this.spin;

        if (this.y > height + 20) {

          this.reset(false);

        }

      }


      draw() {

        ctx.save();

        ctx.translate(this.x, this.y);

        ctx.rotate(this.angle);

        ctx.globalAlpha = this.opacity;

        ctx.fillStyle = this.color;


        if (this.type === 'petal') {

          ctx.beginPath();

          ctx.moveTo(0, -this.size / 2);

          ctx.bezierCurveTo(this.size / 2, -this.size / 2, this.size / 2, this.size / 2, 0, this.size / 2);

          ctx.bezierCurveTo(-this.size / 2, this.size / 2, -this.size / 2, -this.size / 2, 0, -this.size / 2);

          ctx.fill();

        } else {

          const s = this.size * 0.65;

          ctx.beginPath();

          ctx.moveTo(0, s / 4);

          ctx.bezierCurveTo(0, 0, -s, 0, -s, s / 2);

          ctx.bezierCurveTo(-s, s, 0, s * 1.35, 0, s * 1.6);

          ctx.bezierCurveTo(0, s * 1.35, s, s, s, s / 2);

          ctx.bezierCurveTo(s, 0, 0, 0, 0, s / 4);

          ctx.fill();

        }

        ctx.restore();

      }

    }


    for (let i = 0; i < particleCount; i++) {

      particles.push(new Particle());

    }


    function animateParticles() {

      ctx.clearRect(0, 0, width, height);

      particles.forEach(p => {

        p.update();

        p.draw();

      });

      requestAnimationFrame(animateParticles);

    }

    let noteIndex = 0;


    function playSoftNote(freq) {

      if (!audioCtx) return;

      const now = audioCtx.currentTime;

      const osc = audioCtx.createOscillator();

      const gain = audioCtx.createGain();


      osc.type = 'sine';

      osc.frequency.setValueAtTime(freq, now);


      gain.gain.setValueAtTime(0.0001, now);

      gain.gain.exponentialRampToValueAtTime(0.06, now + 0.12);

      gain.gain.exponentialRampToValueAtTime(0.0001, now + 2.2);


      osc.connect(gain);

      gain.connect(audioCtx.destination);


      osc.start(now);

      osc.stop(now + 2.3);

    }


    function toggleMusic() {

          audioCtx = new (window.AudioContext || window.webkitAudioContext)();

          audioCtx.resume();

        }

        musicBtn.classList.add('active');

        musicLabel.textContent = 'Musiqani to‘xtatish';

        musicIcon.textContent = '❚❚';


        playSoftNote(notes[noteIndex]);

        noteIndex = (noteIndex + 1) % notes.length;


        melodyTimer = setInterval(() => {

          playSoftNote(notes[noteIndex]);

          noteIndex = (noteIndex + 1) % notes.length;

        }, 680);

      } else {

        isPlaying = false;

        clearInterval(melodyTimer);

        musicBtn.classList.remove('active');

        musicLabel.textContent = 'Musiqani yoqish';

        musicIcon.textContent = '♪';

      }

    }
